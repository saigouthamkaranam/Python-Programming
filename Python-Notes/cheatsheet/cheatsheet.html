Property decorators
class BankAccount:
    def __init__(self, balance):
        self.__balance = balance   # private

    @property
    def balance(self):             # getter
        return self.__balance

    @balance.setter
    def balance(self, v):           # setter
        if v < 0: raise ValueError
        self.__balance = v

acc = BankAccount(1000)
acc.balance = 2000              # setter called

Generators

Intermediate
Generator functions
def count_up(max):
    n = 1
    while n <= max:
        yield n          # pause here
        n += 1

for n in count_up(5):
    print(n)

# Or use next() manually
gen = count_up(3)
next(gen)   # 1
next(gen)   # 2

Decorators

Intermediate
Basic decorator
from functools import wraps

def timer(func):
    @wraps(func)            # preserve metadata
    def wrapper(*args, **kwargs):
        import time
        start = time.time()
        result = func(*args, **kwargs)
        print(f"{time.time()-start:.4f}s")
        return result
    return wrapper

@timer
def slow_fn():
    print("running...")
Decorator with arguments
def repeat(n):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kw):
            for _ in range(n):
                func(*args, **kw)
        return wrapper
    return decorator

@repeat(3)
def hello():
    print("Hello!")

hello()   # prints "Hello!" 3 times

Error Handling

Intermediate
try / except / else / finally
try:
    result = 10 / 0
except ZeroDivisionError as e:
    print(f"Error: {e}")
except (TypeError, ValueError):
    print("Type or value error")
except Exception as e:       # catch-all
    print(e)
else:
    print("No error!")         # runs if no exception
finally:
    print("Always runs")       # cleanup
Custom exceptions & raising
class InsufficientFundsError(Exception):
    def __init__(self, amount):
        super().__init__(f"Need {amount} more")

def withdraw(balance, amount):
    if amount > balance:
        raise InsufficientFundsError(amount - balance)
    return balance - amount

try:
    withdraw(100, 200)
except InsufficientFundsError as e:
    print(e)   # Need 100 more

Async / Await

Advanced
Basics
import asyncio

async def fetch_data(url):
    print(f"Fetching {url}")
    await asyncio.sleep(1)    # non-blocking wait
    return {"data": "result"}

async def main():
    result = await fetch_data("api/v1")
    print(result)

asyncio.run(main())
Concurrent tasks
import asyncio

async def task(name, delay):
    await asyncio.sleep(delay)
    print(f"{name} done")

async def main():
    # Run concurrently — not one by one!
    await asyncio.gather(
        task("A", 2),
        task("B", 1),
        task("C", 3),
    )
# Total: ~3s, not 6s
async / sync — key concepts
ConceptMeaning
async defDefines a coroutine
awaitPause until result is ready
asyncio.run()Entry point — runs event loop
asyncio.gather()Run multiple coroutines concurrently
asyncio.sleep()Non-blocking wait
asyncio.create_task()Schedule task without awaiting
Use async for I/O bound tasks — HTTP calls, DB queries, file reads. Use multiprocessing for CPU-heavy tasks.
